# The functions linked in `Proof/github_com/mit_pdos/etcd_grove/examples/linkv3p/Link.lean`
# (all Lean-first), read by `etc/gen-link.py --linkv3p`:
# `<fn>: <callees in the order of <fn>.wp_proof's hypotheses>`. The results are in the namespace
# `linkv3p`. Perennial's standard-library results (`time.Sleep`, `primitive.RandomUint64`) are
# used directly by the proofs, not linked here.
#
# `startMember`: linkv3's start of one member plus its gRPC server (the server's, v3rpc's and the
# client's results, linked in their `Link.lean`s; the feature gate's, `net.Listen`'s,
# `RaftHandler`'s, `http.Serve`'s, grpc's `KeepaliveEnforcementPolicy`'s and `Serve`'s trust-boundary
# axioms)

v3rpc_Server: @
features_NewDefaultServerFeatureGate:
net_Listen:
EtcdServer__RaftHandler:
http_Serve:
grpc_Server__Serve:
grpc_KeepaliveEnforcementPolicy:
linkv3p_startMember: features_NewDefaultServerFeatureGate etcdserver_NewServer EtcdServer__Start net_Listen EtcdServer__RaftHandler http_Serve grpc_KeepaliveEnforcementPolicy v3rpc_Server grpc_Server__Serve
# `newClient`: the client's `New` (linked in the client's `Link.lean`)
client_v3_New: @
linkv3p_newClient: client_v3_New
# `startBridge`: `net.Listen` and the bridge's `New` (the bridge's trust-boundary axioms, D26)
bridge_New:

# `partitioner`: `peerStep` over etcd's partition hooks and the member id (linked in the server's
# `Link.lean`), `clientStep` over the bridge's `Blackhole`, `Unblackhole`, `DropConnections`

EtcdServer__MendPeer: @

bridge__Unblackhole:
bridge__DropConnections:

linkv3p_partitioner: linkv3p_peerStep linkv3p_clientStep
# `Apply` over linkvtxn's; `Check` over the driver's `Run` and `CounterClock`, linkvtxn's `Model`
