package fa

import (
	"bytes"
	"fmt"
	"math/rand"
	"path/filepath"
	"sort"
	"testing"

	bolt "go.etcd.io/bbolt"
)

var bkt = []byte("b")

func key(i int) []byte { return []byte(fmt.Sprintf("k%08d", i)) }

func openBig(t *testing.T, n, vlen int) *bolt.DB {
	t.Helper()
	db, err := bolt.Open(filepath.Join(t.TempDir(), "db"), 0600, &bolt.Options{InitialMmapSize: 1 << 30, NoSync: true})
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { db.Close() })
	if err := db.Update(func(tx *bolt.Tx) error {
		b, err := tx.CreateBucket(bkt)
		if err != nil {
			return err
		}
		for i := 0; i < n; i++ {
			if err := b.Put(key(i), bytes.Repeat([]byte{byte(i)}, vlen)); err != nil {
				return err
			}
		}
		return nil
	}); err != nil {
		t.Fatal(err)
	}
	return db
}

func walk(b *bolt.Bucket) ([]string, error) {
	var out []string
	err := b.ForEach(func(k, v []byte) error {
		if k == nil {
			panic("nil key")
		}
		out = append(out, string(k))
		return nil
	})
	return out, err
}

// Family 4/6: write tx with deletes emptying whole leaves (first, middle, last, whole subtrees)
// and puts into deleted ranges; ForEach (First, not Seek) must walk exactly the model's sorted keys.
func TestWriteTxEmptyLeaves(t *testing.T) {
	for seed := int64(0); seed < 40; seed++ {
		n := 30000
		vlen := 1 + int(seed%5)*40
		db := openBig(t, n, vlen)
		r := rand.New(rand.NewSource(seed))
		model := map[string]bool{}
		for i := 0; i < n; i++ {
			model[string(key(i))] = true
		}
		if err := db.Update(func(tx *bolt.Tx) error {
			b := tx.Bucket(bkt)
			// delete random ranges, including prefix and suffix
			ranges := [][2]int{{0, r.Intn(8000) + 1}, {n - r.Intn(8000) - 1, n}}
			for j := 0; j < 5; j++ {
				a := r.Intn(n)
				ranges = append(ranges, [2]int{a, a + r.Intn(6000)})
			}
			for _, rg := range ranges {
				for i := rg[0]; i < rg[1] && i < n; i++ {
					if err := b.Delete(key(i)); err != nil {
						return err
					}
					delete(model, string(key(i)))
				}
			}
			// some puts back into deleted ranges
			for j := 0; j < 50; j++ {
				i := r.Intn(n)
				if err := b.Put(key(i), []byte("new")); err != nil {
					return err
				}
				model[string(key(i))] = true
			}
			got, err := walk(b)
			if err != nil {
				return err
			}
			var exp []string
			for k := range model {
				exp = append(exp, k)
			}
			sort.Strings(exp)
			if len(got) != len(exp) {
				t.Fatalf("seed %d: got %d entries, model %d", seed, len(got), len(exp))
			}
			for i := range got {
				if got[i] != exp[i] {
					t.Fatalf("seed %d: entry %d: %q vs %q", seed, i, got[i], exp[i])
				}
			}
			return nil
		}); err != nil {
			t.Fatal(err)
		}
	}
}

// Delete every key in a write tx, then ForEach: no visit.
func TestWriteTxAllDeleted(t *testing.T) {
	db := openBig(t, 20000, 100)
	db.Update(func(tx *bolt.Tx) error {
		b := tx.Bucket(bkt)
		for i := 0; i < 20000; i++ {
			b.Delete(key(i))
		}
		got, _ := walk(b)
		if len(got) != 0 {
			t.Fatalf("got %d", len(got))
		}
		// and a single put in the middle
		b.Put(key(12345), nil)
		got, _ = walk(b)
		if len(got) != 1 || got[0] != string(key(12345)) {
			t.Fatalf("got %v", got)
		}
		return nil
	})
}

type stash struct {
	k, v   []byte
	kc, vc []byte
}

func check(t *testing.T, where string, st []stash) {
	for i, s := range st {
		if !bytes.Equal(s.k, s.kc) || !bytes.Equal(s.v, s.vc) {
			t.Fatalf("%s: stashed slice %d changed: %q/%q", where, i, s.k, s.kc)
		}
	}
}

// Family 3, write tx: stash the visitor's slices, then mutate the tx through the contract
// (Put overwriting, Delete with the stashed key, DeleteBucket, CreateBucketIfNotExists of the same
// name, puts into the new bucket); the stashed bytes must not change while the tx is open.
func TestWriteTxStashSurvivesMutation(t *testing.T) {
	for _, vlen := range []int{0, 10, 3000, 70000} {
		db := openBig(t, 3000, vlen)
		db.Update(func(tx *bolt.Tx) error {
			b := tx.Bucket(bkt)
			// materialize some leaves with own writes first
			for i := 0; i < 3000; i += 7 {
				b.Put(key(i), bytes.Repeat([]byte{0xAA}, vlen))
			}
			var st []stash
			b.ForEach(func(k, v []byte) error {
				st = append(st, stash{k, v, append([]byte{}, k...), append([]byte{}, v...)})
				return nil
			})
			for i := 0; i < 3000; i += 3 {
				b.Put(key(i), bytes.Repeat([]byte{0x55}, vlen+1))
			}
			check(t, "after puts", st)
			for i := 0; i < len(st); i += 2 {
				b.Delete(st[i].k) // Delete_kv own=false
			}
			check(t, "after deletes", st)
			tx.DeleteBucket(bkt)
			check(t, "after DeleteBucket", st)
			nb, _ := tx.CreateBucketIfNotExists(bkt)
			for i := 0; i < 5000; i++ {
				nb.Put(key(i), bytes.Repeat([]byte{0x11}, vlen))
			}
			check(t, "after recreate+puts", st)
			return nil
		})
	}
}

// Family 3, read tx: stash slices, then a concurrent writer overwrites/deletes/regrows and commits
// many times while the read tx is open; the stashed bytes must not change.
func TestReadTxStashSurvivesCommits(t *testing.T) {
	db := openBig(t, 3000, 500)
	tx, err := db.Begin(false)
	if err != nil {
		t.Fatal(err)
	}
	var st []stash
	tx.Bucket(bkt).ForEach(func(k, v []byte) error {
		st = append(st, stash{k, v, append([]byte{}, k...), append([]byte{}, v...)})
		return nil
	})
	done := make(chan error)
	go func() {
		for round := 0; round < 30; round++ {
			err := db.Update(func(w *bolt.Tx) error {
				b := w.Bucket(bkt)
				for i := 0; i < 3000; i++ {
					if (i+round)%2 == 0 {
						b.Delete(key(i))
					} else {
						b.Put(key(i), bytes.Repeat([]byte{byte(round)}, 500))
					}
				}
				return nil
			})
			if err != nil {
				done <- err
				return
			}
		}
		done <- nil
	}()
	if err := <-done; err != nil {
		t.Fatal(err)
	}
	check(t, "read tx after commits", st)
	tx.Rollback()
}

// Family 4: empty and nil values: ForEach still visits them with a non-nil key (the loop tests k).
func TestEmptyValues(t *testing.T) {
	db := openBig(t, 0, 0)
	db.Update(func(tx *bolt.Tx) error {
		b := tx.Bucket(bkt)
		b.Put([]byte("a"), nil)
		b.Put([]byte("b"), []byte{})
		n := 0
		b.ForEach(func(k, v []byte) error { n++; t.Logf("w: k=%q v=%v nil=%v", k, v, v == nil); return nil })
		if n != 2 {
			t.Fatalf("write: %d", n)
		}
		return nil
	})
	db.View(func(tx *bolt.Tx) error {
		n := 0
		tx.Bucket(bkt).ForEach(func(k, v []byte) error { n++; t.Logf("r: k=%q v=%v nil=%v", k, v, v == nil); return nil })
		if n != 2 {
			t.Fatalf("read: %d", n)
		}
		return nil
	})
}

// Family 1 (kvr): a cursor made before the walk (ownCur in I) used by the visitor: harmless?
func TestReadVisitorUsesPriorCursor(t *testing.T) {
	db := openBig(t, 5000, 50)
	db.View(func(tx *bolt.Tx) error {
		b := tx.Bucket(bkt)
		c := b.Cursor()
		c.Seek(key(0))
		n := 0
		b.ForEach(func(k, v []byte) error {
			k2, _ := c.Next()
			c.Seek(k)
			_ = k2
			n++
			return nil
		})
		if n != 5000 {
			t.Fatalf("%d", n)
		}
		return nil
	})
}
