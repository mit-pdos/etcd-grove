// Does Go's default http.Serve (the http boundary's Serve_spec) answer an HTTP/2 prior-knowledge
// client (grpc-go's transport) with HTTP/2? If not, an HTTP server registered at a grpc address
// cannot hand a grpc client a successful reply.
package main

import (
	"fmt"
	"io"
	"net"
	"net/http"
	"time"
)

func main() {
	l, err := net.Listen("tcp", "127.0.0.1:0")
	if err != nil {
		panic(err)
	}
	go http.Serve(l, http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		fmt.Println("handler ran:", r.Proto, r.Method, r.URL)
		w.Header().Set("content-type", "application/grpc")
		w.Write([]byte{0, 0, 0, 0, 0})
	}))
	c, err := net.Dial("tcp", l.Addr().String())
	if err != nil {
		panic(err)
	}
	// client preface + empty SETTINGS frame
	c.Write([]byte("PRI * HTTP/2.0\r\n\r\nSM\r\n\r\n"))
	c.Write([]byte{0, 0, 0, 4, 0, 0, 0, 0, 0})
	c.SetReadDeadline(time.Now().Add(2 * time.Second))
	b, _ := io.ReadAll(c)
	fmt.Printf("%q\n", b)
}
